"use client";

import type {
  DraggableAttributes,
  DraggableSyntheticListeners,
} from "@dnd-kit/core";
import {
  addMinutes,
  differenceInMinutes,
  format,
  getMinutes,
  isPast,
} from "date-fns";

import { cn } from "@/lib/utils";
import { Button } from "@/components/ui/button";

import type { CalendarEvent, CalendarView } from "../types";
import { eventColorClasses, getBorderRadiusClasses } from "../utils";

function formatTime(date: Date) {
  return format(date, getMinutes(date) === 0 ? "ha" : "h:mma").toLowerCase();
}

interface EventItemProps {
  event: CalendarEvent;
  view: CalendarView;
  isDragging?: boolean;
  onClick?: (e: React.MouseEvent) => void;
  showTime?: boolean;
  currentTime?: Date;
  isFirstDay?: boolean;
  isLastDay?: boolean;
  showTitle?: boolean;
  dndListeners?: DraggableSyntheticListeners;
  dndAttributes?: DraggableAttributes;
}

export function EventItem({
  event,
  view,
  isDragging,
  onClick,
  showTime,
  currentTime,
  isFirstDay = true,
  isLastDay = true,
  showTitle = true,
  dndListeners,
  dndAttributes,
}: EventItemProps) {
  const start = currentTime ?? event.start;
  const duration = differenceInMinutes(event.end, event.start);
  const end = addMinutes(start, duration);
  const isShort = duration < 45;

  const buttonProps = {
    onClick,
    "data-dragging": isDragging || undefined,
    "data-past-event": isPast(end) || undefined,
    ...dndListeners,
    ...dndAttributes,
  };

  const dot = (className?: string) => (
    <span
      aria-hidden="true"
      className={cn(
        "size-1.5 shrink-0 rounded-full",
        eventColorClasses[event.color],
        className,
      )}
    />
  );

  if (view === "month") {
    return (
      <Button
        {...buttonProps}
        variant="ghost"
        className={cn(
          "bg-muted text-foreground hover:bg-muted/60 data-past-event:text-muted-foreground mt-(--event-gap) flex h-(--event-height) justify-start gap-1.5 px-1.5 text-xs font-medium active:not-aria-[haspopup]:translate-y-0 data-dragging:cursor-grabbing data-dragging:shadow-lg",
          "@max-xl/month:bg-transparent @max-xl/month:px-0 @max-xl/month:hover:bg-transparent",
          getBorderRadiusClasses(isFirstDay, isLastDay),
          isFirstDay && "ms-1",
          isFirstDay && isLastDay && "w-[calc(100%-0.5rem)]",
          isFirstDay !== isLastDay && "w-[calc(100%-0.25rem)]",
          !isFirstDay && !isLastDay && "w-full",
        )}
      >
        {dot(
          cn(
            "@max-xl/month:h-1.5 @max-xl/month:w-full",
            !isFirstDay && "hidden @max-xl/month:block",
          ),
        )}
        <span
          className={cn(
            "truncate group-data-past-event/button:line-through @max-xl/month:sr-only",
            !showTitle && "invisible",
          )}
        >
          {!event.allDay && isFirstDay && (
            <span className="text-muted-foreground font-normal">
              {formatTime(start)}{" "}
            </span>
          )}
          {event.title}
        </span>
      </Button>
    );
  }

  if (view === "week" || view === "day") {
    return (
      <Button
        {...buttonProps}
        variant="ghost"
        className={cn(
          "bg-muted text-foreground border-background hover:bg-muted/70 data-past-event:text-muted-foreground relative h-full w-full justify-start overflow-hidden rounded-md border py-1 ps-3 pe-1.5 text-left text-xs font-medium whitespace-normal active:not-aria-[haspopup]:translate-y-0 data-dragging:cursor-grabbing data-dragging:shadow-lg data-past-event:line-through",
          isShort ? "items-center gap-1" : "flex-col items-start gap-0",
        )}
      >
        <span
          aria-hidden="true"
          className={cn(
            "absolute inset-y-1 start-1 w-1 rounded-full",
            eventColorClasses[event.color],
          )}
        />
        {isShort ? (
          <span className="truncate">
            {event.title}{" "}
            {showTime && (
              <span className="text-muted-foreground font-normal">
                {formatTime(start)}
              </span>
            )}
          </span>
        ) : (
          <>
            <span className="w-full truncate">{event.title}</span>
            {showTime && (
              <span className="text-muted-foreground w-full truncate font-normal">
                {formatTime(start)} - {formatTime(end)}
              </span>
            )}
          </>
        )}
      </Button>
    );
  }

  return (
    <Button
      {...buttonProps}
      variant="outline"
      className="data-past-event:text-muted-foreground h-auto w-full flex-col items-start gap-1 rounded-xl px-4 py-3 text-left font-normal whitespace-normal active:not-aria-[haspopup]:translate-y-0"
    >
      <span className="flex w-full items-center gap-2 text-sm font-medium group-data-past-event/button:line-through">
        {dot("size-2")}
        <span className="truncate">{event.title}</span>
      </span>
      <span className="text-muted-foreground text-xs">
        {event.allDay ? "All day" : `${formatTime(start)} - ${formatTime(end)}`}
        {event.location && ` · ${event.location}`}
      </span>
      {event.description && (
        <span className="text-muted-foreground text-xs">
          {event.description}
        </span>
      )}
    </Button>
  );
}
