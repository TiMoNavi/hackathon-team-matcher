```python
import requests

def get_github_profile(username):
    """
    获取 GitHub 用户的技能画像
    """
    # 基础信息
    user = requests.get(f'https://api.github.com/users/{username}').json()
    
    # 仓库列表
    repos = requests.get(f'https://api.github.com/users/{username}/repos').json()
    
    # 分析编程语言分布
    languages = {}
    for repo in repos:
        lang = repo.get('language')
        if lang:
            languages[lang] = languages.get(lang, 0) + 1
    
    return {
        'name': user['name'],
        'bio': user['bio'],
        'company': user['company'],
        'location': user['location'],
        'followers': user['followers'],
        'public_repos': user['public_repos'],
        'languages': languages,
    }
```

```typescript
// lib/github.ts
import { Octokit } from '@octokit/rest'

const octokit = new Octokit({
  auth: process.env.GITHUB_TOKEN
})

export async function analyzeGitHubProfile(username: string) {
  try {
    // 1. 基础信息
    const { data: user } = await octokit.users.getByUsername({ username })
    
    // 2. 仓库列表
    const { data: repos } = await octokit.repos.listForUser({
      username,
      sort: 'updated',
      per_page: 100
    })
    
    // 3. 分析编程语言
    const languages: Record<string, number> = {}
    for (const repo of repos) {
      if (repo.language) {
        languages[repo.language] = (languages[repo.language] || 0) + 1
      }
    }
    
    // 4. 计算能力评分
    const activityScore = Math.min(100, user.public_repos * 2)
    const influenceScore = Math.min(100, Math.log10(user.followers + 1) * 20)
    
    return {
      username: user.login,
      name: user.name,
      bio: user.bio,
      location: user.location,
      languages,
      stats: {
        repos: user.public_repos,
        followers: user.followers,
        activityScore,
        influenceScore
      },
      verified: true,  // 来自 GitHub 的数据置信度高
    }
  } catch (error) {
    console.error('GitHub API error:', error)
    return null
  }
}
```

## 四、数据验证方案

### 4.1 自述数据的验证规则

#### 技能验证矩阵

| 自述技能 | 验证来源 | 验证规则 | 置信度调整 |
|----------|----------|----------|------------|
| "精通 React" | GitHub repos | 是否有 3+ 个 React 项目 | 有: +30%, 无: -40% |
| "5 年经验" | LinkedIn | 工作时间线是否匹配 | 匹配: +20%, 不匹配: -50% |
| "拿过奖" | 证明材料 | 上传获奖证书 | 有: +40%, 无: 0% |
| "全栈工程师" | GitHub | 前后端代码都有 | 都有: +30%, 只有一端: -20% |

#### 验证流程
```typescript
async function verifyUserSkills(userId: string) {
  const user = await prisma.user.findUnique({
    where: { id: userId },
    include: { skills: true }
  })
  
  const verifications = []
  
  for (const skill of user.skills) {
    let confidence = 0.5  // 初始置信度
    
    // 1. GitHub 验证
    if (user.githubUrl) {
      const githubData = await analyzeGitHubProfile(user.githubUrl)
      if (githubData.languages[skill.skillName]) {
        confidence += 0.3
      } else {
        confidence -= 0.2
      }
    }
    
    // 2. 项目经验验证
    const relevantProjects = user.projects.filter(p =>
      p.techStack.includes(skill.skillName)
    )
    if (relevantProjects.length > 0) {
      confidence += 0.2
    }
    
    // 3. 技能等级与经验年限一致性
    const expectedLevel = Math.min(5, Math.floor(user.experienceYears / 2) + 2)
    if (Math.abs(skill.skillLevel - expectedLevel) > 2) {
      confidence -= 0.3  // 声称的等级与经验不符
    }
    
    verifications.push({
      skillId: skill.id,
      confidence: Math.max(0, Math.min(1, confidence)),
      verified: confidence > 0.6
    })
  }
  
  // 更新数据库
  for (const v of verifications) {
    await prisma.userSkill.update({
      where: { id: v.skillId },
      data: { 
        verified: v.verified,
        confidence: v.confidence
      }
    })
  }
}
```

### 4.2 交叉验证策略

#### 多源数据对比

```typescript
interface VerificationSources {
  self_reported: UserProfile      // 用户自述
  github: GitHubProfile | null    // GitHub 数据
  linkedin: LinkedInProfile | null // LinkedIn 数据
  peer_reviews: PeerReview[]      // 历史互评
}

function crossValidate(sources: VerificationSources) {
  const scores = {
    self_reported: 0.4,  // 权重 40%
    github: 0.3,         // 权重 30%
    linkedin: 0.2,       // 权重 20%
    peer_reviews: 0.1,   // 权重 10%
  }
  
  // 如果数据源缺失，自动调整权重
  const available = Object.keys(sources).filter(k => sources[k])
  const totalWeight = available.reduce((sum, k) => sum + scores[k], 0)
  
  // 加权平均
  const finalScore = available.reduce((sum, source) => {
    const weight = scores[source] / totalWeight
    const sourceScore = calculateSourceScore(sources[source])
    return sum + weight * sourceScore
  }, 0)
  
  return finalScore
}
```

```typescript
interface PeerReview {
  reviewerId: string
  revieweeId: string
  teamId: string
  ratings: {
    skill_accuracy: number    // 技能描述准确度 1-5
    technical_ability: number // 实际技术能力 1-5
    communication: number     // 沟通能力 1-5
    collaboration: number     // 协作态度 1-5
  }
  comments: string
}

async function adjustScoreBasedOnReview(review: PeerReview) {
  const user = await prisma.user.findUnique({
    where: { id: review.revieweeId }
  })
  
  // 如果技能准确度评分低，下调技能置信度
  if (review.ratings.skill_accuracy <= 2) {
    await prisma.user.update({
      where: { id: user.id },
      data: {
        credibilityScore: user.credibilityScore - 10,
        // 下调所有技能的置信度
      }
    })
  }
  
  // 如果评分高，提升信誉分
  if (review.ratings.skill_accuracy >= 4) {
    await prisma.user.update({
      where: { id: user.id },
      data: {
        credibilityScore: user.credibilityScore + 5
      }
    })
  }
}
```

```typescript
function anonymizeResume(resume: any) {
  return {
    ...resume,
    name: hash(resume.name),          // 哈希化姓名
    email: mask(resume.email),         // 脱敏邮箱
    phone: mask(resume.phone),         // 脱敏手机号
    address: resume.address?.city,     // 只保留城市
  }
}

function mask(str: string) {
  if (!str) return ''
  const half = Math.floor(str.length / 2)
  return str.slice(0, half) + '*'.repeat(str.length - half)
}
```

```typescript
async function calculateDataQuality(userId: string) {
  const user = await prisma.user.findUnique({
    where: { id: userId },
    include: { skills: true, projects: true }
  })
  
  // 完整度
  const totalFields = 10
  const filledFields = [
    user.nickname,
    user.contactEmail || user.contactWechat,
    user.skills.length > 0,
    user.projects.length > 0,
    user.githubUrl,
    // ...
  ].filter(Boolean).length
  const completeness = filledFields / totalFields
  
  // 验证率
  const verifiedSkills = user.skills.filter(s => s.verified).length
  const verificationRate = verifiedSkills / user.skills.length
  
  // 一致性（简化示例）
  const consistency = user.credibilityScore / 100
  
  return {
    completeness,
    verificationRate,
    consistency,
    overall: (completeness + verificationRate + consistency) / 3
  }
}
```
